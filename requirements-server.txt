# Backend (FastAPI) dependencies for Privacy Filter - Local.
# The core ML deps (torch, huggingface_hub, safetensors, tiktoken, numpy) come
# from privacy-filter/pyproject.toml (installed via `pip install -e .`).
# Install with:  pip install -r requirements-server.txt
fastapi==0.141.1
uvicorn[standard]==0.52.0
python-multipart==0.0.32
PyMuPDF==1.28.0
python-docx==1.2.0
# OCR fallback for scanned PDFs. Also requires the tesseract binary and the
# 'spa' traineddata to be installed on the host (brew install tesseract
# tesseract-lang; apt install tesseract-ocr tesseract-ocr-spa). Both are
# imported lazily so the app still works without OCR — pages without a text
# layer are just left blank and reported in the warning.
pytesseract==0.3.13
Pillow==12.3.0
# Iberian NER (adds recall on top of the primarily-English opf model). This
# installs the spaCy runtime only. The language models are NOT pinned here on
# purpose: they are ~1.2 GB of data, and server/ner_models.py downloads them
# into the app folder on first run, the same way the opf checkpoint is fetched.
#
# This comment used to say "you also need: python -m spacy download
# es_core_news_lg". Nothing acted on it — not a launcher, not the Windows
# build, not CI — so every release up to 2.6.3 ran without the models and left
# every person name written in caps unredacted. The startup preflight in
# server/inference.py now refuses to make the app available until they are
# installed, and smoke_ner.py fails the release build if they cannot be.
#
# On a development machine `python -m spacy download es_core_news_lg` still
# works: a pip-installed model counts as present, so nothing is re-downloaded,
# and it is loaded as-is.
spacy==3.8.13
# Language detection so we run only the NER model matching each document's
# language and avoid tagging Catalan common words as Spanish person names
# (and vice versa). Very small pure-Python package.
langdetect==1.0.9
# Optional post-processing: convert the *already-redacted* PDF/DOCX/etc. to
# Markdown so the user can paste it into an LLM to summarise or draft with
# fewer tokens and preserved structure.
#
# IMPORTANT: SELECTIVE extras only, NEVER markitdown[all]. Leaving out
# `audio-transcription` (which calls SpeechRecognition.recognize_google and
# uploads audio to Google), `youtube-transcription`, and the Azure clients
# means those network paths are not merely disabled — they are absent from
# the package. tests/test_offline_guarantee.py fails if this ever slips.
# Also `llm_client=None` is set at construction, so images cannot be sent to
# an external model for captioning.
markitdown[pdf,docx,pptx,xlsx,xls,outlook]==0.1.7
# CA bundle for the update checks. Already present transitively (via
# huggingface_hub → requests), pinned here because app_update now verifies
# against it directly: a python.org framework Python on macOS has no CA store
# of its own until the user runs Install Certificates.command, so without this
# every urlopen to GitHub fails with CERTIFICATE_VERIFY_FAILED and the app
# silently never reports an update.
certifi==2026.7.22
